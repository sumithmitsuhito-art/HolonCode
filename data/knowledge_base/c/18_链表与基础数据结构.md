# 模块 18：链表与基础数据结构

## 一、单向链表

```c
#include <stdlib.h>
typedef struct Node {
    int data;
    struct Node *next;
} Node;

/* 头插：在头部插入 */
Node *list_push_front(Node *head, int val) {
    Node *new_node = (Node*)malloc(sizeof(Node));
    if (!new_node) return head;
    new_node->data = val;
    new_node->next = head;
    return new_node;      // 新的头
}

/* 尾插 */
Node *list_push_back(Node *head, int val) {
    Node *n = (Node*)malloc(sizeof(Node));
    if (!n) return head;
    n->data = val; n->next = NULL;
    if (!head) return n;
    Node *p = head;
    while (p->next) p = p->next;
    p->next = n;
    return head;
}

/* 查找 */
Node *list_find(Node *head, int key) {
    for (Node *p = head; p; p = p->next)
        if (p->data == key) return p;
    return NULL;
}

/* 删除（注意：要找到前一个节点） */
Node *list_erase(Node *head, int key) {
    Node dummy;
    dummy.next = head;     // 假头法，简化"删头"的特殊情况
    Node *prev = &dummy;
    while (prev->next) {
        if (prev->next->data == key) {
            Node *target = prev->next;
            prev->next = target->next;
            free(target);
            break;
        }
        prev = prev->next;
    }
    return dummy.next;
}

/* 释放整个链表 */
void list_free(Node *head) {
    while (head) {
        Node *next = head->next;
        free(head);
        head = next;
    }
}
```

## 二、栈（Stack，LIFO） — 先进后出

用数组实现：
```c
typedef struct { int data[1024]; int top; } Stack;
void push(Stack *s, int v) { s->data[s->top++] = v; }
int  pop(Stack *s)          { return s->data[--s->top]; }
int  peek(Stack *s)         { return s->data[s->top - 1]; }
int  empty(Stack *s)        { return s->top == 0; }
```

## 三、队列（Queue，FIFO） — 先进先出

用环形数组：
```c
typedef struct {
    int data[1024];
    int head, tail, size;
} Queue;
void enqueue(Queue *q, int v) {
    q->data[q->tail] = v;
    q->tail = (q->tail + 1) % 1024;
    q->size++;
}
int dequeue(Queue *q) {
    int v = q->data[q->head];
    q->head = (q->head + 1) % 1024;
    q->size--;
    return v;
}
```
